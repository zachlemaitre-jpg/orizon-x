> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `py relay.py --encoder amf` with a double-clickable Windows app (frosted-glass window + tray icon) that runs the relay, bundled as a standalone exe with ffmpeg and adb.

**Architecture:** The relay loop moves from `relay.py` into `engine.py` (`RelayEngine`: a thread, a `Listener` for states/logs/stats, clean `stop()`, errors as states instead of `sys.exit`). `relay.py` becomes a thin CLI over the engine. A PySide6 app (`host-prototype/app/`) drives the same engine through a `Controller` that turns engine callbacks into Qt signals. PyInstaller (one-folder) packages it; `build-pc.ps1` adds ffmpeg/adb and shortcuts.

**Tech Stack:** Python 3.13 (`py`), stdlib `unittest`, PySide6-Essentials (Qt 6), PyInstaller, ctypes (DWM), winreg, PowerShell.

```python
    directory.mkdir(parents=True, exist_ok=True)

class FindToolTest(unittest.TestCase):
    def setUp(self):
        self.root = Path(tempfile.mkdtemp())
        self.bundled = self.root / "tools"
        self.sdk = self.root / "sdk"
        self.on_path = self.root / "path"

    def find(self, name, path_hit=True):
        def which(tool):
            candidate = self.on_path / f"{tool}{EXE}"
            return str(candidate) if path_hit and candidate.exists() else None
        return tools.find_tool(name, bundled_dir=self.bundled, sdk_dir=self.sdk, which=which)
```

```python
"""Finds the external tools (ffmpeg, adb) and runs them without a console window.

In the packaged app they sit in tools/ next to OrizonX.exe; in development they come from the PATH or the
Android SDK.
"""
from __future__ import annotations

import os
import shutil
import subprocess
import sys
from pathlib import Path
from typing import Callable

# Sans ce drapeau, chaque appel à adb ou ffmpeg depuis l'exe graphique ouvre une console qui clignote.
NO_WINDOW = subprocess.CREATE_NO_WINDOW if os.name == "nt" else 0

_EXE = ".exe" if os.name == "nt" else ""

_MISSING = {
    "ffmpeg": "ffmpeg introuvable. Recompile l'appli (build-pc.ps1) ou installe ffmpeg (winget install Gyan.FFmpeg).",
    "adb": "adb introuvable. Recompile l'appli (build-pc.ps1) ou installe Android platform-tools.",
}


class ToolMissing(Exception):
    """A required tool is nowhere to be found; the message tells the user what to do."""


def app_dir() -> Path:
    """Folder of OrizonX.exe when packaged, of this file in development."""
    if getattr(sys, "frozen", False):
        return Path(sys.executable).resolve().parent
    return Path(__file__).resolve().parent


def sdk_platform_tools() -> Path:
    local = os.environ.get("LOCALAPPDATA") or str(Path.home() / "AppData" / "Local")
    return Path(local) / "Android" / "Sdk" / "platform-tools"


def find_tool(name: str, *, bundled_dir: Path | None = None, sdk_dir: Path | None = None,
              which: Callable[[str], str | None] = shutil.which) -> str:
    """Path of ffmpeg or adb, or ToolMissing.

    ffmpeg: the bundled copy first (the version we tested). adb: the installed one first, because two adb versions
    on one PC keep killing each other's server; the bundled copy is the last resort.
    """
    if name not in _MISSING:
        raise ValueError(f"outil inconnu : {name}")
    bundled = (bundled_dir if bundled_dir is not None else app_dir() / "tools") / f"{name}{_EXE}"
    if name == "ffmpeg":
        candidates = [bundled, which(name)]
    else:
        sdk = (sdk_dir if sdk_dir is not None else sdk_platform_tools()) / f"{name}{_EXE}"
        candidates = [which(name), sdk, bundled]
    for candidate in candidates:
        if candidate and Path(candidate).is_file():
            return str(candidate)
    raise ToolMissing(_MISSING[name])
```

```python
"""Relay engine: waits for the tablet over adb, streams one screen through ffmpeg, injects touch and pen.

The CLI (relay.py) and the PC app (app/) both drive it through a Listener. Threads: one engine thread and, per
session, an ffmpeg stderr pump and a tablet reader.
"""
from __future__ import annotations

import collections
import contextlib
import re
import socket
import subprocess
import threading
import time
import traceback
from dataclasses import dataclass
from enum import Enum
from typing import Any, BinaryIO, Callable, Protocol

import annexb
import injector
import protocol
import tools


class State(Enum):
    STOPPED = "stopped"
    WAITING_DEVICE = "waiting_device"  # adb : pas de tablette, non autorisée, plusieurs…
    WAITING_APP = "waiting_app"  # adb OK, Orizon X pas ouverte sur la tablette
    STREAMING = "streaming"
    ERROR = "error"  # fatal : le moteur s'est arrêté


@dataclass(frozen=True)
class Config:
    output: str = "auto"  # "auto" ou un index ddagrab
    encoder: str = "auto"  # "auto" | "nvenc" | "amf"
    fps: int = 60
    bitrate: str = "30M"


@dataclass(frozen=True)
class StreamStats:
    width: int
    height: int
    fps: int
    latency_avg_ms: float
    latency_max_ms: float
    dropped: int
    rtt_ms: float


class Listener(Protocol):
    """Called from the engine's threads: must not block. `detail` is a French sentence ready to show."""

    def on_state(self, state: State, detail: str) -> None: ...

    def on_log(self, line: str) -> None: ...

    def on_stats(self, stats: StreamStats) -> None: ...


class FatalError(Exception):
    """Stops the engine; the message is shown to the user as is."""


ENCODERS = {
    "nvenc": ["-c:v", "h264_nvenc", "-preset", "p1", "-tune", "ull", "-zerolatency", "1", "-delay", "0"],
    # AMF : -usage ultralowlatency/lowlatency ignorent -g (une seule IDR) ; sans -usage, -latency 1 donne la basse latence en gardant le GOP.
    # -aud 0 : sinon AMF ajoute ses propres AUD, en double avec ceux du bsf h264_metadata.
    "amf": ["-c:v", "h264_amf", "-latency", "1", "-quality", "speed", "-aud", "0"],
}
ENCODER_NAMES = {"nvenc": "NVENC", "amf": "AMF"}
ENCODER_CHOICES = ("auto", "nvenc", "amf")
AUTO_ORDER = ("nvenc", "amf")  # NVENC d'abord (sur secteur), AMF sinon (sur batterie, ou pilote NVIDIA trop ancien)

MAX_FFMPEG_FAILURES = 3
RETRY_DELAY_S = 2.0
STOP_TIMEOUT_S = 3.0
ENCODER_PROBE_TIMEOUT_S = 15

WAITING_APP_DETAIL = "Tablette branchée : ouvre Orizon X dessus."

Run = Callable[..., subprocess.CompletedProcess]
Outputs = list[tuple[int, tuple[int, int] | None]]

_ONLY_DEFAULT_GPU = ("Seuls les écrans du GPU par défaut sont listés "
                     "(sur un portable hybride, ceux branchés sur l'autre GPU n'apparaissent pas).")


def now_us() -> int:
    # perf_counter (QueryPerformanceCounter) : time.monotonic n'a que ~15,6 ms de résolution sous Windows avant 3.13.
    return time.perf_counter_ns() // 1000


class SocketReader:
    """Minimal binary reader over recv(). Unlike makefile(), it lets sock.close() abort a blocked recv on Windows."""

    def __init__(self, sock: socket.socket) -> None:
        self._sock = sock

    def read(self, size: int) -> bytes:
        return self._sock.recv(size)


def probe_outputs(ffmpeg: str, max_outputs: int = 8, run: Run = subprocess.run) -> Outputs:
    """Captures one frame per ddagrab output (default D3D11 adapter) and returns (index, (width, height))."""
    outputs: Outputs = []
    for index in range(max_outputs):
        result = run(
            [ffmpeg, "-nostdin", "-hide_banner", "-f", "lavfi",
             "-i", f"ddagrab=output_idx={index}:framerate=1,hwdownload,format=bgra",
             "-frames:v", "1", "-f", "null", "-"],
            capture_output=True, text=True, errors="replace", creationflags=tools.NO_WINDOW,
        )
        if result.returncode != 0:
            break  # les index sont contigus : le premier échec marque la fin
        size = re.search(r"\b(\d{3,5})x(\d{3,5})\b", result.stderr)
        outputs.append((index, (int(size.group(1)), int(size.group(2))) if size else None))
    return outputs


def describe_outputs(outputs: Outputs) -> str:
    if not outputs:
        return "Aucun écran capturable (ffmpeg/ddagrab indisponible ?)"
    lines = [f"--output {i} : {f'{s[0]}x{s[1]}' if s else 'taille inconnue'}" for i, s in outputs]
    return "\n".join(lines + [_ONLY_DEFAULT_GPU])


def pick_output(choice: str, outputs: Outputs, width: int, height: int) -> int:
    """Explicit index, or ('auto') the first screen whose size matches the tablet's."""
    if choice != "auto":
        return int(choice)
    for index, size in outputs:
        if size == (width, height):
            return index
    raise FatalError(f"Aucun écran en {width}x{height} (taille de la tablette). Règle l'écran virtuel à cette "
                     f"résolution, ou choisis un écran dans les réglages :\n{describe_outputs(outputs)}")


def ffmpeg_command(ffmpeg: str, encoder: str, output: int, fps: int, bitrate: str) -> list[str]:
    return [
        ffmpeg, "-nostdin", "-hide_banner", "-loglevel", "warning",
        # hwdownload : sur le portable hybride l'écran est sur l'AMD et NVENC sur la NVIDIA (corrigé en phase 2).
        "-f", "lavfi", "-i", f"ddagrab=output_idx={output}:framerate={fps},hwdownload,format=bgra",
        *ENCODERS[encoder],
        "-rc", "cbr", "-b:v", bitrate, "-bf", "0", "-g", str(fps),
        "-bsf:v", "h264_metadata=aud=insert", "-f", "h264", "-",
    ]


def probe_encoder(ffmpeg: str, encoder: str, run: Run = subprocess.run) -> bool:
    """True if this hardware encoder can encode one frame right now (driver present, GPU awake)."""
    try:
        result = run(
            [ffmpeg, "-nostdin", "-hide_banner", "-loglevel", "error",
             "-f", "lavfi", "-i", "color=c=black:s=256x256:r=1", "-frames:v", "1",
             *ENCODERS[encoder], "-f", "null", "-"],
            capture_output=True, creationflags=tools.NO_WINDOW, timeout=ENCODER_PROBE_TIMEOUT_S,
        )
    except subprocess.TimeoutExpired:
        return False
    return result.returncode == 0


def resolve_encoder(ffmpeg: str, choice: str, run: Run = subprocess.run) -> tuple[str, str]:
    """(encoder, log line). 'auto' probes NVENC then AMF; a forced encoder is trusted as is."""
    if choice != "auto":
        return choice, f"Encodeur : {ENCODER_NAMES[choice]} (forcé)"
    unavailable: list[str] = []
    for name in AUTO_ORDER:
        if probe_encoder(ffmpeg, name, run):
            suffix = f" ({', '.join(unavailable)} indisponible)" if unavailable else ""
            return name, f"Encodeur : {ENCODER_NAMES[name]}{suffix}"
        unavailable.append(ENCODER_NAMES[name])
    raise FatalError("Aucun encodeur matériel H.264 (AMF/NVENC) disponible.")


def adb_problem(adb: str, run: Run = subprocess.run) -> str | None:
    """None if exactly one authorized device is ready, else a message saying what to do."""
    result = run([adb, "devices"], capture_output=True, text=True, creationflags=tools.NO_WINDOW)
    if result.returncode != 0:
        return f"adb ne répond pas : {result.stderr.strip()}"
    states = [line.split("\t")[1] for line in result.stdout.splitlines()[1:] if "\t" in line]
    if not states:
        return "Aucune tablette détectée : branche le câble USB et active le débogage USB."
    if "unauthorized" in states:
        return "Tablette non autorisée : accepte la demande de débogage USB sur l'écran de la tablette."
    if states.count("device") != 1:
        return f"Il faut exactement une tablette prête. État actuel : {states}"
    return None


def ensure_forward(adb: str, run: Run = subprocess.run) -> str | None:
    """(Re)creates the adb forward; returns a message if it can't. adb drops forwards when the device reconnects."""
    problem = adb_problem(adb, run)
    if problem is not None:
        return problem
    result = run([adb, "forward", f"tcp:{protocol.PORT}", f"tcp:{protocol.PORT}"],
                 capture_output=True, text=True, creationflags=tools.NO_WINDOW)
    if result.returncode != 0:
        return f"adb forward a échoué : {result.stderr.strip()}"
    return None


def format_stats(stats: StreamStats) -> str:
    return (f"{stats.fps:3d} fps | latence moy {stats.latency_avg_ms:5.1f} ms, "
            f"max {stats.latency_max_ms:5.1f} ms | perdues {stats.dropped} | RTT {stats.rtt_ms:.2f} ms")


def stream(source: BinaryIO, send: Callable[[bytes], None], fps: int, stop: threading.Event,
           on_size: Callable[[int, int], None], log: Callable[[str], None]) -> None:
    """Forwards ffmpeg's Annex-B output as TABX messages until EOF or stop."""
    splitter = annexb.AccessUnitSplitter()
    size: tuple[int, int] | None = None
    last_sps: bytes | None = None
    last_pps: bytes | None = None
    bad_sps: bytes | None = None
    while not stop.is_set():
        chunk = source.read1(65536)
        if not chunk:
            return
        for unit in splitter.feed(chunk):
            info = annexb.describe(unit)
            if size is None and not (info.is_idr and info.sps and info.pps):
                continue  # on attend la première IDR avec ses SPS/PPS
            if info.sps is not None and info.pps is not None and (info.sps != last_sps or info.pps != last_pps):
                try:
                    new_size = annexb.parse_sps_size(info.sps)
                except ValueError as e:
                    if info.sps != bad_sps:
                        log(f"SPS illisible ({e}) : unité ignorée")
                        bad_sps = info.sps
                    continue
                if new_size != size:
                    size = new_size
                    on_size(size[0], size[1])
                    send(protocol.encode_config(size[0], size[1], fps))
                    log(f"Flux : {size[0]}x{size[1]} @ {fps} fps")
                codec_config = annexb.START_CODE + info.sps + annexb.START_CODE + info.pps
                send(protocol.encode_video(now_us(), codec_config, protocol.FLAG_CODEC_CONFIG))
                last_sps = info.sps
                last_pps = info.pps
            if info.is_idr and info.sps is None and last_sps is not None:
                unit = annexb.insert_parameter_sets(unit, last_sps, last_pps)
            send(protocol.encode_video(now_us(), unit, protocol.FLAG_KEY if info.is_idr else 0))
```

```python
class FakeTablet:
    """The tablet end of a socket pair: sends HELLO at once and records every frame the PC sends."""

    def __init__(self, version=protocol.VERSION):
        self.pc, self.tab = socket.socketpair()
        self.frames = queue.Queue()
        self.tab.sendall(hello_frame(version))
        threading.Thread(target=self._drain, daemon=True).start()

    def _drain(self):
        reader = self.tab.makefile("rb")
        try:
            while True:
                head = reader.read(protocol.HEADER.size)
                if len(head) < protocol.HEADER.size:
                    break
                msg_type, flags, _reserved, length = protocol.HEADER.unpack(head)
                self.frames.put((msg_type, flags, reader.read(length)))
        except (OSError, ValueError):
            pass
        self.frames.put(None)

    def next_message(self, of_type, timeout=5.0):
        deadline = time.monotonic() + timeout
        while True:
            frame = self.frames.get(timeout=max(0.01, deadline - time.monotonic()))
            if frame is None:
                raise AssertionError("connexion fermée par le PC")
            if frame[0] == of_type:
                return frame

    def wait_closed(self, timeout=5.0):
        deadline = time.monotonic() + timeout
        while True:
            if self.frames.get(timeout=max(0.01, deadline - time.monotonic())) is None:
                return

    def send(self, data):
        self.tab.sendall(data)

    def leave(self):
        self.tab.shutdown(socket.SHUT_RDWR)

class Recorder:
    def __init__(self):
        self.states = []
        self.logs = []
        self.stats = []
        self._cond = threading.Condition()

    def on_state(self, state, detail):
        with self._cond:
            self.states.append((state, detail))
            self._cond.notify_all()

    def on_log(self, line):
        with self._cond:
            self.logs.append(line)
            self._cond.notify_all()

    def on_stats(self, stats):
        with self._cond:
            self.stats.append(stats)
            self._cond.notify_all()

    def wait_for(self, predicate, timeout=5.0):
        with self._cond:
            if not self._cond.wait_for(predicate, timeout):
                raise AssertionError(f"délai dépassé ; états : {self.states}")

    def wait_state(self, state, timeout=5.0):
        self.wait_for(lambda: any(s is state for s, _ in self.states), timeout)

    def last_state(self):
        return self.states[-1][0] if self.states else None
```

```python
class Deps:
    """Everything the engine touches outside Python; tests swap in fakes."""
    run: Run = subprocess.run

class SessionResult:
    ffmpeg_died: bool  # ffmpeg s'est arrêté seul (ni la tablette ni stop())
    sent_video: bool
    stderr_tail: list[str]


class RelayEngine:
    """Runs the relay loop in its own thread until stop() or a fatal error (state ERROR)."""

    def __init__(self, config: Config, listener: Listener, *, ffmpeg: str, adb: str,
                 deps: Deps | None = None) -> None:
        self._config = config
        self._listener = listener
        self._ffmpeg = ffmpeg
        self._adb = adb
        self._deps = deps or Deps()
        self._lock = threading.Lock()
        self._stop_event = threading.Event()
        self._thread: threading.Thread | None = None
        self._active_sock: socket.socket | None = None
        self._active_ffmpeg: Any = None
        self._state = State.STOPPED
        self._detail = ""
        self._last_encoder_line: str | None = None

    @property
    def state(self) -> State:
        return self._state

    def is_running(self) -> bool:
        thread = self._thread
        return thread is not None and thread.is_alive()

    def start(self) -> None:
        """Starts the loop; does nothing if it is already running."""
        with self._lock:
            if self.is_running():
                return
            self._stop_event = threading.Event()
            self._thread = threading.Thread(target=self._run, name="relay-engine", daemon=True)
            self._thread.start()

    def stop(self) -> None:
        """Blocks up to STOP_TIMEOUT_S: kills ffmpeg, closes the socket, lifts every touch contact."""
        with self._lock:
            self._stop_event.set()
            thread = self._thread
            # Fermer le socket et tuer ffmpeg débloque tous les threads de la session.
            if self._active_sock is not None:
                with contextlib.suppress(OSError):
                    self._active_sock.close()
            if self._active_ffmpeg is not None:
                with contextlib.suppress(OSError):
                    self._active_ffmpeg.kill()
        if thread is not None and thread is not threading.current_thread():
            thread.join(STOP_TIMEOUT_S)

    def wait(self, timeout: float | None = None) -> bool:
        """True once the engine thread has ended."""
        thread = self._thread
        if thread is None:
            return True
        thread.join(timeout)
        return not thread.is_alive()

    # --- thread du moteur ---

    def _run(self) -> None:
        try:
            self._loop()
        except FatalError as e:
            self._set_state(State.ERROR, str(e))
            return
        except Exception as e:  # noqa: BLE001 — l'appli ne doit jamais planter : on montre l'erreur
            self._log(traceback.format_exc().rstrip())
            self._set_state(State.ERROR, f"Erreur inattendue : {e}")
            return
        finally:
            with self._lock:
                self._active_sock = None
                self._active_ffmpeg = None
        self._set_state(State.STOPPED, "Arrêté.")

    def _loop(self) -> None:
        failures = 0
        while not self._stop_event.is_set():
            encoder = self._pick_encoder()
            # Sondage avant la connexion : une fois le HELLO reçu, la tablette attend des PONG (5 s de silence = coupure).
            outputs = probe_outputs(self._ffmpeg, run=self._deps.run) if self._config.output == "auto" else []
            connection = self._connect()
            if connection is None:
                return
            sock, rfile, hello = connection
            self._log(f"Tablette connectée : écran {hello.width}x{hello.height}, {hello.dpi} dpi")
            try:
                output = pick_output(self._config.output, outputs, hello.width, hello.height)
            except FatalError:
                self._close_socket(sock)
                raise
            if self._config.output == "auto":
                self._log(f"Écran capturé : --output {output} ({hello.width}x{hello.height}, choisi automatiquement)")
            result = self._session(sock, rfile, hello, encoder, output)
            if self._stop_event.is_set():
                return
            if result.ffmpeg_died:
                self._log("ffmpeg s'est arrêté de lui-même. Dernières lignes :\n  " + "\n  ".join(result.stderr_tail))
                failures = 0 if result.sent_video else failures + 1
                if failures >= MAX_FFMPEG_FAILURES:
                    last = result.stderr_tail[-1] if result.stderr_tail else "aucun message"
                    raise FatalError(f"ffmpeg s'arrête sans cesse ({failures} fois de suite). Dernier message : {last}")
                self._stop_event.wait(self._deps.retry_delay_s)
            else:
                failures = 0
            self._log("Nouvelle tentative de connexion…")

    def _pick_encoder(self) -> str:
        encoder, line = resolve_encoder(self._ffmpeg, self._config.encoder, self._deps.run)
        if line != self._last_encoder_line:
            self._log(line)
            self._last_encoder_line = line
        return encoder

    def _connect(self) -> tuple[socket.socket, SocketReader, protocol.Hello] | None:
        """Waits for a tablet, redoing the adb forward on each try, then connects and waits for HELLO."""
        while not self._stop_event.is_set():
            problem = ensure_forward(self._adb, self._deps.run)
            if problem is not None:
                self._set_state(State.WAITING_DEVICE, problem)
                self._stop_event.wait(self._deps.poll_s)
                continue
            try:
                sock = self._deps.create_connection(("127.0.0.1", protocol.PORT), timeout=2)
            except OSError:
                self._wait_for_app()
                continue
            with self._lock:
                if self._stop_event.is_set():
                    sock.close()
                    return None
                self._active_sock = sock
            try:
                with contextlib.suppress(OSError):
                    sock.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
                # adb accepte la connexion locale même si rien n'écoute sur la tablette : seul le HELLO prouve que l'app est là.
                rfile = SocketReader(sock)
                hello = protocol.read_message(rfile)
                sock.settimeout(None)
            except (OSError, EOFError, protocol.ProtocolError):
                self._close_socket(sock)
                if self._stop_event.is_set():
                    return None
                self._wait_for_app()
                continue
            if not isinstance(hello, protocol.Hello):
                self._close_socket(sock)
                raise FatalError(f"Premier message inattendu de la tablette : {hello}")
            if hello.version != protocol.VERSION:
                self._close_socket(sock)
                raise FatalError(f"Protocole incompatible : tablette v{hello.version}, PC v{protocol.VERSION}. "
                                 "Mets à jour l'app ou le relais.")
            return sock, rfile, hello
        return None

    def _wait_for_app(self) -> None:
        self._set_state(State.WAITING_APP, WAITING_APP_DETAIL)
        self._stop_event.wait(self._deps.poll_s)

    def _session(self, sock: socket.socket, rfile: SocketReader, hello: protocol.Hello, encoder: str,
                 output: int) -> SessionResult:
        """Streams until the tablet leaves, stop() is called, or ffmpeg exits on its own."""
        send_lock = threading.Lock()
        sent_video = threading.Event()

        def send(data: bytes) -> None:
            with send_lock:
                sock.sendall(data)
            sent_video.set()  # stream() n'envoie rien avant la première IDR : tout envoi est de la vidéo

        def send_pong(t_tab_us: int) -> None:
            with send_lock:  # l'horodatage est pris sous le verrou : pas d'attente entre la mesure et l'envoi
                sock.sendall(protocol.encode_pong(t_tab_us, now_us()))

        command = ffmpeg_command(self._ffmpeg, encoder, output, self._config.fps, self._config.bitrate)
        ffmpeg = self._deps.popen(command, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE,
                                  stderr=subprocess.PIPE, creationflags=tools.NO_WINDOW)
        with self._lock:
            self._active_ffmpeg = ffmpeg
            if self._stop_event.is_set():
                ffmpeg.kill()
        self._set_state(State.STREAMING, "Démarrage du flux…")
        stderr_tail: collections.deque[str] = collections.deque(maxlen=20)
        disconnected = threading.Event()
        closing = threading.Event()  # posé quand le relais ferme lui-même le socket : pas de message d'erreur parasite
        keyframe_requests = 0  # la tablette en envoie une juste après chaque CONFIG (normal) : on ne journalise que les suivantes
        input_failed = False  # un échec d'injection n'est journalisé qu'une fois par session
        stream_size: tuple[int, int] | None = None
        pointer_injector: Any = None
        injector_lock = threading.Lock()

        def on_stream_size(width: int, height: int) -> None:
            nonlocal pointer_injector, stream_size
            stream_size = (width, height)
            self._set_state(State.STREAMING, f"Flux {width}×{height}")
            with injector_lock:
                if pointer_injector is not None:
                    self._close_injector(pointer_injector)
                    pointer_injector = None
                try:
                    pointer_injector = self._deps.create_injector(width, height, self._log)
                except Exception as e:  # noqa: BLE001 — le tactile ne doit jamais interrompre la vidéo
                    self._log(f"Tactile désactivé : {e}. La vidéo continue.")
                    pointer_injector = None

        def pump_stderr() -> None:
            try:
                for line in ffmpeg.stderr:
                    text = line.decode(errors="replace").rstrip()
                    stderr_tail.append(text)
                    self._log(f"[ffmpeg] {text}")
            except (OSError, ValueError):
                pass  # le thread ne doit jamais mourir tant que ffmpeg écrit

        def read_tablet() -> None:
            nonlocal keyframe_requests, input_failed
            try:
                while True:
                    message = protocol.read_message(rfile)
                    if isinstance(message, protocol.Ping):
                        send_pong(message.t_tab_us)
                    elif isinstance(message, protocol.Stats):
                        size = stream_size
                        if size is not None:
                            self._listener.on_stats(StreamStats(
                                size[0], size[1], message.fps, message.latency_avg_us / 1000,
                                message.latency_max_us / 1000, message.dropped, message.rtt_us / 1000))
                    elif isinstance(message, protocol.KeyframeRequest):
                        keyframe_requests += 1
                        if keyframe_requests >= 2:
                            self._log("Keyframe demandée par la tablette : prochaine IDR sous 1 s (GOP fixe)")
                    elif isinstance(message, protocol.Input):
                        with injector_lock:
                            target = pointer_injector
                        if target is not None:
                            try:
                                target.handle(message)
                            except Exception as e:  # noqa: BLE001 — le tactile ne doit jamais interrompre la vidéo
                                if not input_failed:
                                    input_failed = True
                                    self._log(f"Tactile en échec, entrées ignorées : {e}")
            except (EOFError, OSError, protocol.ProtocolError) as e:
                if not closing.is_set() and not self._stop_event.is_set():
                    self._log(f"Tablette déconnectée ({e})")
            finally:
                disconnected.set()
                with contextlib.suppress(OSError):
                    sock.close()  # sous Windows, seul closesocket() interrompt un sendall bloqué dans stream()
                with contextlib.suppress(OSError):
                    ffmpeg.kill()

        stderr_thread = threading.Thread(target=pump_stderr, name="ffmpeg-stderr", daemon=True)
        stderr_thread.start()
        threading.Thread(target=read_tablet, name="tablet-reader", daemon=True).start()
        send_failed = False
        try:
            stream(ffmpeg.stdout, send, self._config.fps, disconnected, on_stream_size, self._log)
        except OSError:
            send_failed = True  # sendall a échoué : la tablette est partie
        finally:
            with injector_lock:
                if pointer_injector is not None:
                    self._close_injector(pointer_injector)  # lève tous les contacts : rien ne reste « posé » dans Windows
                    pointer_injector = None
            tablet_gone = send_failed or disconnected.is_set()
            with contextlib.suppress(OSError):
                ffmpeg.kill()
            ffmpeg.wait()
            closing.set()
            self._close_socket(sock)
            with self._lock:
                self._active_ffmpeg = None
        died = not tablet_gone and not self._stop_event.is_set()
        if died:
            stderr_thread.join(timeout=0.5)  # laisse le pump finir de lire avant de rendre la fin du stderr
        return SessionResult(died, sent_video.is_set(), list(stderr_tail))

    # --- utilitaires ---

    def _close_socket(self, sock: socket.socket) -> None:
        with self._lock:
            if self._active_sock is sock:
                self._active_sock = None
        with contextlib.suppress(OSError):
            sock.close()

    def _close_injector(self, target: Any) -> None:
        try:
            target.close()
        except Exception as e:  # noqa: BLE001 — ne doit jamais empêcher la fermeture de la session
            self._log(f"Fermeture du tactile en échec : {e}")

    def _set_state(self, state: State, detail: str) -> None:
        if (state, detail) == (self._state, self._detail):
            return
        self._state = state
        self._detail = detail
        self._listener.on_state(state, detail)

    def _log(self, line: str) -> None:
        self._listener.on_log(line)
```

```python
"""Relay CLI (debug mode): the same engine as the Orizon X PC app, logging to the terminal.

from __future__ import annotations

import argparse
import sys

import engine
import injector
import tools


def output_choice(value: str) -> str:
    if value == "auto" or value.isdigit():
        return value
    raise argparse.ArgumentTypeError("attendu : 'auto' ou un numéro d'écran")


def parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Relais Orizon X : écran PC -> tablette via adb")
    parser.add_argument("--encoder", choices=engine.ENCODER_CHOICES, default="auto",

    parser.add_argument("--bitrate", default="30M")
    parser.add_argument("--list-outputs", action="store_true", help="liste les écrans capturables puis quitte")

    """Engine listener that prints everything, like the phase-1 relay did."""

    def on_state(self, state: engine.State, detail: str) -> None:
        print(detail)

    def on_log(self, line: str) -> None:

    def on_stats(self, stats: engine.StreamStats) -> None:
        print(engine.format_stats(stats))
```

```python
from __future__ import annotations

import json
import os
from dataclasses import asdict, dataclass
from pathlib import Path

ENCODERS = ("auto", "nvenc", "amf")


@dataclass
class Settings:
    output: str = "auto"  # "auto" ou un index ddagrab
    encoder: str = "auto"
    first_close_hint_shown: bool = False  # bulle « Orizon X tourne toujours » déjà montrée


def default_path() -> Path:
    roaming = os.environ.get("APPDATA") or str(Path.home() / "AppData" / "Roaming")
    return Path(roaming) / "OrizonX" / "settings.json"


def load(path: Path) -> Settings:
    """Never fails: a missing or broken file gives the defaults, a bad field only resets that field."""
    try:
        raw = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return Settings()
    result = Settings()
    if not isinstance(raw, dict):
        return result
    output = raw.get("output")
    if isinstance(output, str) and (output == "auto" or output.isdigit()):
        result.output = output
    encoder = raw.get("encoder")
    if encoder in ENCODERS:
        result.encoder = encoder
    hint = raw.get("first_close_hint_shown")
    if isinstance(hint, bool):
        result.first_close_hint_shown = hint
    return result


def save(settings: Settings, path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_name(path.name + ".tmp")
    temp.write_text(json.dumps(asdict(settings), indent=2), encoding="utf-8")
    os.replace(temp, path)  # atomique : jamais de fichier à moitié écrit
```

```python
"""'Lancer avec Windows': the HKCU Run value is the single source of truth."""
from __future__ import annotations

import sys

RUN_KEY = r"Software\Microsoft\Windows\CurrentVersion\Run"
VALUE_NAME = "OrizonX"


def supported() -> bool:
    """Only the packaged exe can be registered (a source checkout has no stable executable)."""
    return bool(getattr(sys, "frozen", False))


def command_for(exe: str) -> str:
    return f'"{exe}" --minimized'


def registered_command(key_path: str = RUN_KEY) -> str | None:
    import winreg
    try:
        with winreg.OpenKey(winreg.HKEY_CURRENT_USER, key_path) as key:
            value, _kind = winreg.QueryValueEx(key, VALUE_NAME)
            return value
    except FileNotFoundError:
        return None


def is_enabled(key_path: str = RUN_KEY) -> bool:
    return registered_command(key_path) is not None


def enable(exe: str, key_path: str = RUN_KEY) -> None:
    import winreg
    with winreg.CreateKey(winreg.HKEY_CURRENT_USER, key_path) as key:
        winreg.SetValueEx(key, VALUE_NAME, 0, winreg.REG_SZ, command_for(exe))


def disable(key_path: str = RUN_KEY) -> None:
    import winreg
    try:
        with winreg.OpenKey(winreg.HKEY_CURRENT_USER, key_path, 0, winreg.KEY_SET_VALUE) as key:
            winreg.DeleteValue(key, VALUE_NAME)
    except FileNotFoundError:
        pass
```

```python
"""Bridges the relay engine (its own threads) to Qt (GUI thread) and owns start, stop and restart."""
from __future__ import annotations

import threading
from typing import Callable, Protocol

from PySide6.QtCore import QObject, Signal

from engine import Config, Listener, State, StreamStats


class Engine(Protocol):
    def start(self) -> None: ...

    def stop(self) -> None: ...

    def is_running(self) -> bool: ...


EngineFactory = Callable[[Config, Listener], Engine]


class Controller(QObject):
    """Engine Listener whose callbacks are re-emitted as Qt signals.

    Signals emitted from engine threads are queued to the GUI thread, so slots never touch widgets off-thread.
    """

    state_changed = Signal(object, str)
    log = Signal(str)
    stats = Signal(object)

    def __init__(self, factory: EngineFactory, config: Config, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._factory = factory
        self._config = config
        self._engine: Engine | None = None
        self._lock = threading.Lock()
        self.state = State.STOPPED
        self.detail = "Arrêté."
        self.state_changed.connect(self._remember)

    @property
    def config(self) -> Config:
        return self._config

    def _remember(self, state: State, detail: str) -> None:
        self.state = state
        self.detail = detail

    # --- Listener (appelé depuis les threads du moteur) ---

    def on_state(self, state: State, detail: str) -> None:
        self.state_changed.emit(state, detail)

    def on_log(self, line: str) -> None:
        self.log.emit(line)

    def on_stats(self, stats: StreamStats) -> None:
        self.stats.emit(stats)

    # --- commandes (thread Qt) ---

    def start(self) -> None:
        with self._lock:
            self._start_locked()

    def stop(self) -> threading.Thread | None:
        """Non-blocking: the engine's stop (up to 3 s) runs in a worker thread, returned for tests."""
        with self._lock:
            engine, self._engine = self._engine, None
        if engine is None:
            return None
        worker = threading.Thread(target=engine.stop, name="engine-stop", daemon=True)
        worker.start()
        return worker

    def stop_blocking(self) -> None:
        """For Quit: returns once every touch contact is lifted and ffmpeg is gone."""
        with self._lock:
            engine, self._engine = self._engine, None
        if engine is not None:
            engine.stop()

    def restart(self, config: Config) -> threading.Thread:
        def work() -> None:
            with self._lock:
                old, self._engine = self._engine, None
                self._config = config
                if old is not None:
                    old.stop()
                self._start_locked()

        worker = threading.Thread(target=work, name="engine-restart", daemon=True)
        worker.start()
        return worker

    def _start_locked(self) -> None:
        if self._engine is not None and self._engine.is_running():
            return
        try:
            self._engine = self._factory(self._config, self)
        except Exception as e:  # noqa: BLE001 — ex. ToolMissing : on l'affiche au lieu de planter
            self._engine = None
            self.on_state(State.ERROR, str(e))
            return
        self._engine.start()
```

```python
"""Builds app/assets/orizonx.png (256 px) and orizonx.ico (16-256 px) from extended/1x/icon_orizonx.png.

Run by build-pc.ps1; the outputs are committed so the app also runs from source without a build.
Usage: .venv\\Scripts\\python.exe app\\make_icons.py

from __future__ import annotations

import os
import struct
import sys
from pathlib import Path

os.environ.setdefault("QT_QPA_PLATFORM", "offscreen")

from PySide6.QtCore import QBuffer, QByteArray, QIODevice, QRectF, Qt  # noqa: E402
from PySide6.QtGui import QGuiApplication, QImage, QPainter, QPainterPath  # noqa: E402

ROOT = Path(__file__).resolve().parents[2]
SOURCE = ROOT / "extended" / "1x" / "icon_orizonx.png"
ASSETS = Path(__file__).resolve().parent / "assets"
SIZES = (16, 20, 24, 32, 40, 48, 64, 128, 256)
CORNER_RATIO = 0.22  # coins arrondis façon icône d'app Windows 11


def rounded(source: QImage, size: int) -> QImage:
    image = QImage(size, size, QImage.Format.Format_ARGB32_Premultiplied)
    image.fill(Qt.GlobalColor.transparent)
    painter = QPainter(image)
    painter.setRenderHint(QPainter.RenderHint.Antialiasing)
    painter.setRenderHint(QPainter.RenderHint.SmoothPixmapTransform)
    path = QPainterPath()
    path.addRoundedRect(QRectF(0, 0, size, size), size * CORNER_RATIO, size * CORNER_RATIO)
    painter.setClipPath(path)
    painter.drawImage(QRectF(0, 0, size, size), source)
    painter.end()
    return image


def png_bytes(image: QImage) -> bytes:
    data = QByteArray()
    buffer = QBuffer(data)
    buffer.open(QIODevice.OpenModeFlag.WriteOnly)
    image.save(buffer, "PNG")
    buffer.close()
    return bytes(data)


def write_ico(path: Path, images: list[tuple[int, bytes]]) -> None:
    """ICO with PNG-compressed entries (supported since Windows Vista)."""
    header = struct.pack("<HHH", 0, 1, len(images))
    offset = len(header) + 16 * len(images)
    entries = b""
    payload = b""
    for size, png in images:
        # 0 dans l'en-tête signifie 256 px
        entries += struct.pack("<BBBBHHII", size % 256, size % 256, 0, 0, 1, 32, len(png), offset + len(payload))
        payload += png
    path.write_bytes(header + entries + payload)


def main() -> int:
    _app = QGuiApplication.instance() or QGuiApplication(sys.argv[:1])
    source = QImage(str(SOURCE))
    if source.isNull():
        print(f"logo introuvable : {SOURCE}", file=sys.stderr)
        return 1
    ASSETS.mkdir(exist_ok=True)
    rounded(source, 256).save(str(ASSETS / "orizonx.png"))
    write_ico(ASSETS / "orizonx.ico", [(size, png_bytes(rounded(source, size))) for size in SIZES])
    print(f"icônes écrites dans {ASSETS}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Windows 11 frosted glass (DWM Acrylic backdrop) for a frameless Qt window."""
from __future__ import annotations

import ctypes
import sys
from ctypes import wintypes

DWMWA_USE_IMMERSIVE_DARK_MODE = 20
DWMWA_WINDOW_CORNER_PREFERENCE = 33
DWMWA_SYSTEMBACKDROP_TYPE = 38
DWMSBT_TRANSIENTWINDOW = 3  # Acrylic
DWMWCP_ROUND = 2
MIN_BUILD = 22621  # Windows 11 22H2 : premier build avec DWMWA_SYSTEMBACKDROP_TYPE


class MARGINS(ctypes.Structure):
    _fields_ = [("cxLeftWidth", ctypes.c_int), ("cxRightWidth", ctypes.c_int),
                ("cyTopHeight", ctypes.c_int), ("cyBottomHeight", ctypes.c_int)]


def _set_attribute(dwmapi, hwnd: int, attribute: int, value: int) -> bool:
    data = ctypes.c_int(value)
    return dwmapi.DwmSetWindowAttribute(wintypes.HWND(hwnd), attribute, ctypes.byref(data), ctypes.sizeof(data)) == 0


def apply_glass(hwnd: int) -> bool:
    """True when the Acrylic backdrop is on; False means the window must paint its own glass."""
    if sys.platform != "win32" or sys.getwindowsversion().build < MIN_BUILD:
        return False
    try:
        dwmapi = ctypes.WinDLL("dwmapi")
    except OSError:
        return False
    _set_attribute(dwmapi, hwnd, DWMWA_USE_IMMERSIVE_DARK_MODE, 1)
    _set_attribute(dwmapi, hwnd, DWMWA_WINDOW_CORNER_PREFERENCE, DWMWCP_ROUND)
    margins = MARGINS(-1, -1, -1, -1)
    if dwmapi.DwmExtendFrameIntoClientArea(wintypes.HWND(hwnd), ctypes.byref(margins)) != 0:
        return False
    return _set_attribute(dwmapi, hwnd, DWMWA_SYSTEMBACKDROP_TYPE, DWMSBT_TRANSIENTWINDOW)
```

```python
def badge_icon(logo: QPixmap, color: str) -> QIcon:
    """The logo with a coloured dot in the bottom-right corner."""
    pixmap = logo.scaled(64, 64, Qt.AspectRatioMode.KeepAspectRatio, Qt.TransformationMode.SmoothTransformation)
    painter = QPainter(pixmap)
    painter.setRenderHint(QPainter.RenderHint.Antialiasing)
    painter.setPen(QPen(QColor(10, 10, 14), 4))
    painter.setBrush(QColor(color))
    painter.drawEllipse(QRectF(36, 36, 26, 26))
    painter.end()
    return QIcon(pixmap)


class Tray(QObject):
    show_requested = Signal()
    toggle_requested = Signal()
    open_log_requested = Signal()
    autostart_toggled = Signal(bool)
    quit_requested = Signal()

    def __init__(self, logo: QPixmap, *, autostart_supported: bool, autostart_enabled: bool) -> None:
        super().__init__()
        self._logo = logo
        self._icon = QSystemTrayIcon()
        self._menu = QMenu()
        self._status = self._menu.addAction("")
        self._status.setEnabled(False)
        self._menu.addSeparator()
        self._toggle = self._add("Arrêter", self.toggle_requested.emit)
        self._add("Ouvrir la fenêtre", self.show_requested.emit)
        self._add("Ouvrir le journal", self.open_log_requested.emit)
        self._autostart = self._menu.addAction("Lancer avec Windows")
        self._autostart.setCheckable(True)
        self._autostart.setChecked(autostart_enabled)
        self._autostart.setEnabled(autostart_supported)
        if not autostart_supported:
            self._autostart.setToolTip("Disponible dans l'appli compilée (OrizonX.exe)")
        self._autostart.toggled.connect(self.autostart_toggled.emit)
        self._menu.addSeparator()
        self._add("Quitter", self.quit_requested.emit)
        self._icon.setContextMenu(self._menu)
        self._icon.activated.connect(self._on_activated)
        self.set_state(State.STOPPED, "Arrêté.")
        self._icon.show()

    def _add(self, text: str, slot) -> QAction:
        action = self._menu.addAction(text)
        action.triggered.connect(slot)
        return action

    def _on_activated(self, reason: QSystemTrayIcon.ActivationReason) -> None:
        if reason == QSystemTrayIcon.ActivationReason.Trigger:
            self.show_requested.emit()

    def set_state(self, state: State, detail: str) -> None:
        title = status.TITLES[state]
        self._status.setText(f"{title} — {status.detail_for(state, detail)}"[:90])
        self._icon.setToolTip(f"Orizon X — {title}\n{status.detail_for(state, detail)}"[:120])
        self._icon.setIcon(badge_icon(self._logo, status.COLORS[state]))
        self._toggle.setText(status.primary_label(state))

    def set_autostart_checked(self, checked: bool) -> None:
        self._autostart.blockSignals(True)
        self._autostart.setChecked(checked)
        self._autostart.blockSignals(False)

    def show_message(self, title: str, text: str) -> None:
        self._icon.showMessage(title, text, QSystemTrayIcon.MessageIcon.Information, 5000)

    def hide(self) -> None:
        self._icon.hide()

    # --- accès pour les tests ---

    def toggle_text(self) -> str:
        return self._toggle.text()

    def status_text(self) -> str:
        return self._status.text()

    def tooltip(self) -> str:
        return self._icon.toolTip()

    def autostart_action_enabled(self) -> bool:
        return self._autostart.isEnabled()

    def trigger(self, text: str) -> None:
        for action in self._menu.actions():
            if action.text() == text:
                action.trigger()
                return
        raise KeyError(text)
```

```python
"""Orizon X PC app: single instance, log file, window and tray around the relay engine."""
from __future__ import annotations

import argparse
import logging
import os
import sys
import threading
from logging.handlers import RotatingFileHandler
from pathlib import Path

from PySide6.QtCore import QObject, Signal
from PySide6.QtGui import QIcon, QPixmap
from PySide6.QtNetwork import QLocalServer, QLocalSocket
from PySide6.QtWidgets import QApplication

import engine
import tools
from app import autostart
from app import settings as settings_store
from app.controller import Controller
from app.glass import apply_glass
from app.tray import Tray
from app.window import MainWindow

ASSETS = Path(__file__).resolve().parent / "assets"
INSTANCE_NAME = f"OrizonX-PC-{os.environ.get('USERNAME', 'user')}"
logger = logging.getLogger("orizonx")


def setup_logging() -> Path:
    directory = Path(os.environ.get("LOCALAPPDATA") or str(Path.home())) / "OrizonX"
    directory.mkdir(parents=True, exist_ok=True)
    path = directory / "relay.log"
    handler = RotatingFileHandler(path, maxBytes=1_000_000, backupCount=3, encoding="utf-8")
    handler.setFormatter(logging.Formatter("%(asctime)s %(message)s"))
    logger.setLevel(logging.INFO)
    logger.addHandler(handler)
    sys.excepthook = lambda kind, value, tb: logger.error("Exception non gérée", exc_info=(kind, value, tb))
    return path


def notify_running_instance() -> bool:
    """True if another Orizon X is running; it is asked to show its window."""
    socket = QLocalSocket()
    socket.connectToServer(INSTANCE_NAME)
    if not socket.waitForConnected(500):
        return False
    socket.write(b"show")
    socket.flush()
    socket.waitForBytesWritten(500)
    socket.disconnectFromServer()
    return True


def make_engine(config: engine.Config, listener: engine.Listener) -> engine.RelayEngine:
    # Outils cherchés à chaque démarrage : « Réessayer » marche après une installation de ffmpeg/adb.
    return engine.RelayEngine(config, listener, ffmpeg=tools.find_tool("ffmpeg"), adb=tools.find_tool("adb"))


class OutputProber(QObject):
    """Lists the capturable screens in the background (ffmpeg takes ~1 s per screen)."""

    done = Signal(object)

    def probe(self) -> None:
        def work() -> None:
            try:
                outputs = engine.probe_outputs(tools.find_tool("ffmpeg"))
            except Exception:  # noqa: BLE001 — liste vide : seul « Auto » reste proposé
                logger.exception("Liste des écrans impossible")
                outputs = []
            self.done.emit(outputs)

        threading.Thread(target=work, name="probe-outputs", daemon=True).start()


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="OrizonX")
    parser.add_argument("--minimized", action="store_true", help="démarre dans la zone de notification")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)

    app = QApplication(sys.argv[:1])
    app.setApplicationName("Orizon X")
    app.setQuitOnLastWindowClosed(False)  # seul « Quitter » ferme l'appli
    if notify_running_instance():
        return 0
    server = QLocalServer()
    QLocalServer.removeServer(INSTANCE_NAME)  # nettoie un nom resté après un plantage
    server.listen(INSTANCE_NAME)

    log_path = setup_logging()
    logger.info("Orizon X démarre")
    logo = QPixmap(str(ASSETS / "orizonx.png"))
    app.setWindowIcon(QIcon(str(ASSETS / "orizonx.ico")))

    settings_path = settings_store.default_path()
    current = settings_store.load(settings_path)
    controller = Controller(make_engine, engine.Config(output=current.output, encoder=current.encoder))
    window = MainWindow(logo)
    window.set_acrylic(apply_glass(int(window.winId())))
    window.set_encoder(current.encoder)
    window.set_outputs([], current.output)
    tray = Tray(logo, autostart_supported=autostart.supported(),
                autostart_enabled=autostart.supported() and autostart.is_enabled())
    prober = OutputProber()

    def show_window() -> None:
        window.show_and_raise()

    def on_state(state: engine.State, detail: str) -> None:
        logger.info("État : %s — %s", state.value, detail)
        window.set_state(state, detail)
        tray.set_state(state, detail)

    def apply_settings() -> None:
        settings_store.save(current, settings_path)
        controller.restart(engine.Config(output=current.output, encoder=current.encoder))

    def on_output(value: str) -> None:
        current.output = value
        apply_settings()

    def on_encoder(value: str) -> None:
        current.encoder = value
        apply_settings()

    def on_toggle() -> None:
        if controller.state in (engine.State.STOPPED, engine.State.ERROR):
            controller.start()
        else:
            controller.stop()

    def on_closed_to_tray() -> None:
        if not current.first_close_hint_shown:
            tray.show_message("Orizon X tourne toujours",
                              "Le flux continue. Clique sur l'icône Orizon X pour rouvrir la fenêtre.")
            current.first_close_hint_shown = True
            settings_store.save(current, settings_path)

    def on_autostart(enabled: bool) -> None:
        try:
            if enabled:
                autostart.enable(sys.executable)
            else:
                autostart.disable()
        except OSError:
            logger.exception("Lancer avec Windows : échec")
        tray.set_autostart_checked(autostart.is_enabled())

    def on_quit() -> None:
        logger.info("Arrêt demandé")
        controller.stop_blocking()  # lève les contacts et tue ffmpeg avant de partir
        tray.hide()
        app.quit()

    def on_new_connection() -> None:
        connection = server.nextPendingConnection()
        connection.readyRead.connect(show_window)
        connection.disconnected.connect(connection.deleteLater)

    controller.state_changed.connect(on_state)
    controller.log.connect(logger.info)
    controller.stats.connect(window.set_stats)
    window.output_changed.connect(on_output)
    window.encoder_changed.connect(on_encoder)
    window.primary_clicked.connect(on_toggle)
    window.closed_to_tray.connect(on_closed_to_tray)
    window.shown.connect(prober.probe)
    prober.done.connect(lambda outputs: window.set_outputs(outputs, current.output))
    tray.show_requested.connect(show_window)
    tray.toggle_requested.connect(on_toggle)
    tray.open_log_requested.connect(lambda: os.startfile(log_path))
    tray.autostart_toggled.connect(on_autostart)
    tray.quit_requested.connect(on_quit)
    server.newConnection.connect(on_new_connection)

    controller.start()
    if args.minimized:
        prober.probe()
    else:
        show_window()
    return app.exec()
```

### Task 11: `build-pc.ps1` — standalone exe with ffmpeg and adb

**Files:**

- [ ] **Step 1: Write the script** — `build-pc.ps1`:

```powershell
<#
.SYNOPSIS
Compile l'appli PC Orizon X dans dist\OrizonX-PC (OrizonX.exe + ffmpeg + adb).

.PARAMETER Shortcuts
Crée aussi les raccourcis « Orizon X » dans le menu Démarrer et sur le bureau.

.EXAMPLE
.\build-pc.ps1 -Shortcuts
#>
param([switch]$Shortcuts)

$ErrorActionPreference = "Stop"
$root = $PSScriptRoot
$hostDir = Join-Path $root "host-prototype"
$venvPython = Join-Path $hostDir ".venv\Scripts\python.exe"
$out = Join-Path $root "dist\OrizonX-PC"
$staging = Join-Path $root "build\pyinstaller-dist"

if (Get-Process OrizonX -ErrorAction SilentlyContinue) {
    throw "Orizon X tourne : quitte-la (icône de notification > Quitter) avant de recompiler."
}

if (-not (Test-Path $venvPython)) {
    Write-Host "Création de l'environnement Python…"
    py -m venv (Join-Path $hostDir ".venv")
    if ($LASTEXITCODE) { throw "py -m venv a échoué" }
}
& $venvPython -m pip install --disable-pip-version-check -q -r (Join-Path $hostDir "requirements-dev.txt")
if ($LASTEXITCODE) { throw "pip install a échoué" }

Push-Location $hostDir
try {
    & $venvPython app\make_icons.py
    if ($LASTEXITCODE) { throw "génération des icônes en échec" }
    & $venvPython -m PyInstaller --noconfirm --clean --onedir --windowed --name OrizonX `
        --icon (Join-Path $hostDir "app\assets\orizonx.ico") `
        --add-data "$(Join-Path $hostDir 'app\assets');app\assets" `
        --distpath $staging `
        --workpath (Join-Path $root "build\pyinstaller-work") `
        --specpath (Join-Path $root "build") `
        (Join-Path $hostDir "orizonx_pc.py")
    if ($LASTEXITCODE) { throw "PyInstaller a échoué" }
} finally {
    Pop-Location
}

if (Test-Path $out) { Remove-Item -Recurse -Force $out }
Move-Item (Join-Path $staging "OrizonX") $out

$toolsDir = New-Item -ItemType Directory -Force (Join-Path $out "tools")
$ffmpeg = (Get-Command ffmpeg -ErrorAction Stop).Source
Copy-Item $ffmpeg $toolsDir
$ffmpegRoot = Split-Path (Split-Path $ffmpeg)
foreach ($name in "LICENSE", "README.txt") {
    $file = Join-Path $ffmpegRoot $name
    if (Test-Path $file) { Copy-Item $file (Join-Path $toolsDir "ffmpeg-$name") }
}
$adbCommand = Get-Command adb -ErrorAction SilentlyContinue
$adbDir = if ($adbCommand) { Split-Path $adbCommand.Source } else { Join-Path $env:LOCALAPPDATA "Android\Sdk\platform-tools" }
foreach ($name in "adb.exe", "AdbWinApi.dll", "AdbWinUsbApi.dll") {
    Copy-Item (Join-Path $adbDir $name) $toolsDir
}
$notice = Join-Path $adbDir "NOTICE.txt"
if (Test-Path $notice) { Copy-Item $notice (Join-Path $toolsDir "adb-NOTICE.txt") }

if ($Shortcuts) {
    $shell = New-Object -ComObject WScript.Shell
    $exe = Join-Path $out "OrizonX.exe"
    foreach ($folder in [Environment]::GetFolderPath("Programs"), [Environment]::GetFolderPath("Desktop")) {
        $link = $shell.CreateShortcut((Join-Path $folder "Orizon X.lnk"))
        $link.TargetPath = $exe
        $link.WorkingDirectory = $out
        $link.IconLocation = "$exe,0"
        $link.Description = "Tablette en second écran"
        $link.Save()
    }
    Write-Host "Raccourcis créés (menu Démarrer et bureau)."
}

Write-Host "OK : $out\OrizonX.exe"
```
